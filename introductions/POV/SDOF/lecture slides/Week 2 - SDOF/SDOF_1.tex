\section{Equation of motion}
\begin{frame}[c]{A bit of context}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item<1-> Mass-spring-damper - the simplest mechanical system of some practical interest...
				\item<2-> We want to derive an equation that describes the motion of the mass, given some IC and/or externally applied forcing.
				\item<3-> Newton's second law states that,
					\begin{equation}
						F=ma \quad\quad 	\sum_i F_i = m\ddot{x}
					\end{equation}
				\item<4-> What forces are acting on our mass - \textbf{free body diagram}. 
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{figure}
			\input{Diagrams/massSpringLSmod.tikz}
			\caption{Mass-spring-damper system}

\begin{frame}[c]{Mass-spring-damper system}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item Newton's second law states that,
				\begin{align}\nonumber
					m\ddot{x} &= \sum_i F_i \\
					m \ddot{x} &= +F_e - F_k - F_c 
				\end{align}
				\item $F_e$ is an external force, defined in the direction of positive travel.
				\item $F_k$ and $F_c$ are forces that oppose motion - but what are they? 
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\begin{figure}
			\centering
			\input{Diagrams/freeBody.tikz}
			\caption{Mass-spring-damper system}
		\end{figure}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[c]{Ideal springs - Hooke's law}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item An ideal spring has no mass and generates a force proportion to its elongation $\Delta x$ .  
				\item Force in opposite direction to displacement and linearly related to the elongation of the spring,
				\begin{equation}
					F_k = -kx_1
				\end{equation}
				\item $k$ is called the \textit{spring stiffness} (N/m).
				\item For a two-sided spring the force is equal and opposite in direction,
				\begin{equation}
					F_k = k\overbrace{(x_2-x_1)}^{\Delta x}
				\end{equation}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag.tikz}
				\caption{Fixed linear spring.}
			\end{figure}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag2.tikz}
				\caption{Ideal linear spring.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Ideal springs - linearity}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Spring force always opposes displacement.
				\begin{itemize}
					\item[-] If $\Delta x<0$ then  $F_k>0$
					\item[-] If $\Delta x>0$ then  $F_k<0$
				\end{itemize}
			\item Ideal springs are linear - the spring force and elongation remain linearly related indefinitely,
			\begin{equation}
				F_k = -kx_1 \quad \quad 2F_k = -k\, 2x_1 
			\end{equation}
			\item Real springs are non-linear:
			\begin{itemize} 
				\item[-] The force/displacement relation begins to change for large displacements.
				\item[-] Softening or hardening springs
				\end{itemize}
			\item What about combinations of springs? Two options: series or parallel
		\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag.tikz}
				\caption{Fixed linear spring.}
			\end{figure}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag2.tikz}
				\caption{Ideal linear spring.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Ideal springs - parallel combination}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item In parallel spring have the same displacement at either end.
				\item Force divides itself into $F_{k1}$ and $F_{k2}$,
				\begin{align}
					F_{k1} &= k_1(x_2-x_1)\\
					F_{k2} &= k_2(x_2-x_1)
				\end{align}
			\item Forces adds to total force,
			\begin{align}
				F_k &= F_{k1}+F_{k2} \\
				&=\underbrace{(k_1+k_2)}_{k_t}(x_2-x_1)
			\end{align}
		\item Total stiffness = sum of individual stiffnesses.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag3.tikz}
				\caption{Parallel combined springs.}
			\end{figure}
		\vspace{0.5cm}
			\begin{equation}
				k_t = \sum_i^N k_i
			\end{equation}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Ideal springs - series combination}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item In series springs have the same force through their lengths.
				\begin{align}\label{stiffness1}
					F_{k} &= k_1(x_2-x_1)\\
					F_{k} &= k_2(x_3-x_2)\label{stiffness2}
				\end{align}
			\item From \ref{stiffness1} we get,
			\begin{equation}
				x_2 = \frac{F_k}{k_1} + x_1
			\end{equation}
		\item Substitute into \ref{stiffness2},
		\begin{equation}
			F_k = k_2(x_3 -x_1 -\frac{F_k}{k_1})
		\end{equation}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag4.tikz}
				\caption{Series combined springs.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Ideal springs - series combination}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Substitute into \ref{stiffness2},
				\begin{equation}
					F_k = k_2(x_3 -x_1 -\frac{F_k}{k_1})
				\end{equation}
				\item A little bit of algebra leads to,
				\begin{equation}
					F_k =\underbrace{ \left(\frac{1}{k_1}+\frac{1}{k_2}\right)^{-1} }_{k}(x_3-x_1)
				\end{equation} 
			\item Total stiffness always less than individual stiffnesses.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/dashDiag4.tikz}
				\caption{Series combined springs.}
			\end{figure}
		\vspace{0.5cm}
		\begin{equation}
			k = \left(\sum_i^N \frac{1}{k_i}\right)
		\end{equation}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Ideal damper - viscous dashpot}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item An viscous damper  has no mass and generates a force proportion to the relative velocity $\Delta \dot{x}$ .  
				\item Force in opposite direction and linearly related to the relative velocity of the dashpot,
				\begin{equation}
					F_c = -c\dot{x}_1
				\end{equation}
				\item $c$ is  the \textit{viscous damping coefficient} (Ns/m).
				\item For a two-sided dashpot,
				\begin{equation}
					F_c = c\overbrace{(\dot{x}_2-\dot{x}_1)}^{\Delta \dot{x}}
				\end{equation}
			\item Same equations apply for series and parallel arrangements as for springs.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/springDiag.tikz}
				\caption{Fixed linear dashpot.}
			\end{figure}
			\begin{figure}
				\centering
				\input{Diagrams/springDiag2.tikz}
				\caption{Ideal linear spring.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Spring-dashpot equation of motion}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Consider spring-dashpot with \textit{no mass}.
				\item Newton's second law says,
				\begin{equation}
					F_e + F_k + F_c = 0
				\end{equation}
				\item Substitute element forces into the above,
				\begin{equation}
					F_e -kx -c\dot{x} = 0
				\end{equation} 
				\item Or as a `force balance', 
				\begin{equation}\label{EoM1}
					F_e = kx + c\dot{x}
				\end{equation}
			\item Equation \ref{EoM1} is a first order linear differential equation with constant coefficients. 
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/dashSpringDiag.tikz}
				\caption{Fixed spring-dashpot system.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[c]{Mass-spring-dashpot EoM}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Newton's second law states that,
				\begin{align}\nonumber
					m \ddot{x} &= F_e + F_k + F_c \\
				 &=  F_e -kx -c\dot{x}
				\end{align}
				\item Isolate external force,
				\begin{equation}\label{EoM2}
					F = m\ddot{x} + kx + c\dot{x}
				\end{equation}
			\item  \textit{Second} order linear differential equation with constant coefficients. 
			\item Called \textit{single degree of freedom (DoF)} system
						\begin{itemize}
			\item[-] DoF: number of independent coordinates required to completely describe configuration.
		\end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\centering
				\input{Diagrams/freeBody.tikz}
				\caption{Mass-spring-damper system}
			\end{figure}

		\end{column}
	\end{columns}
\end{frame}


\section{Coordinates}

\begin{frame}[c]{Preferred coordinates}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item<1-> Consider a the same problem, now including the effect of gravity.
				\begin{itemize}
					\item[-] $y$ is the position of the mass relative to its static position $y_0$ ignoring gravity.
					\item[-] $x$ is the position of the mass relative to its static position under gravity.
					\item[-] $x_{st}=y_0-x_0$ is the static displacement due to gravity ($=mg/k$)
				\end{itemize}
				\item EoM with respect to $y_0$,
				\begin{equation}\label{EoM2}
					F_e = m\ddot{y} + c\dot{y} + ky {\,\color{red}+ \, mg}
				\end{equation}
			\item Extra term $mg$ because $y_0$ is not at equilibrium $x_0$, EoM more complex.
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\input{Diagrams/massSpringLSmod_gravity.tikz}
				\caption{Mass-spring-damper system under gravity.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}


\begin{frame}[c]{Preferred coordinates}
	\begin{columns}[T]
		\begin{column}{0.6\textwidth}
			\begin{itemize}
				\item Writing the displacement as $y = x-x_{st}$ (recall that $x$ is relative to $x_0$) the EoM becomes,
				\begin{equation}\label{EoM2}
					F_e = m\ddot{x} + c\dot{x} + k(x-x_{st}) {\,\color{red}+ \, mg}
				\end{equation}
			\item Since $kx_{st} = mg$, EoM reduces to,
			\begin{equation}\label{EoM2}
				F_e = m\ddot{x} + c\dot{x} + kx
			\end{equation}
			\item By using the equilibrium point $x_0$ as our reference point, our EoM gets simpler. 
			\item More generally - \textit{a change of coordinates can be used to simplify the problem!}
			\begin{itemize}
				\item This will be very important once we move onto multi-DoF systems. 
			\end{itemize}
			\end{itemize}
		\end{column}
		\begin{column}{0.4\textwidth}
			\begin{figure}
				\input{Diagrams/massSpringLSmod_gravity.tikz}
				\caption{Mass-spring-damper system under gravity.}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}

\section{Undamped homogenous solution}
\begin{frame}[c]{Homogenous spring-dashpot solution}
	\begin{itemize}
		\item<1-> Two types of solution to EoM: homogenous (unforced) and particular (forced)
		\item<2-> General solution is obtained as the sum of homogenous and particular.
		\item<3-> \textit{Unforced} spring-dashpot EoM:
		\begin{equation}\label{}
			0 = kx + c\dot{x}
		\end{equation}
		\item<4-> We are interested in solutions of the form $x(t) = Ae^{st}$. Substitute this  into above.
		\begin{equation}\label{}
			0 = kAe^{st} + c \underbrace{sAe^{st}}_{\dot{x}}
		\end{equation}
	\item<5-> Dividing both sides by $Ae^{st}$ we obtain so-called \textit{characteristic equation},
	\begin{equation}
		0 = k + sc \quad \quad s = -\frac{k}{c}
	\end{equation}
	\end{itemize}
\end{frame}

\begin{frame}[c]{Homogenous spring-dashpot solution}
	\begin{itemize}
		\item<1-> Characteristic equation from before,
		\begin{equation}
			0 = k + sc \quad \quad s = -\frac{k}{c}
		\end{equation}
		\item<2-> Using the above, the homogenous solution is $x(t) = Ae^{st} = Ae^{-\frac{k}{c}t}=Ae^{-\frac{t}{\tau}}$, where $\tau=c/k$ is called the `time constant'.
		\item<3-> Suppose at $t=0$ the system is displaced $x(0) = x_0$, we see that,
		\begin{equation}
			x(0) = Ae^{-\frac{0}{\tau}} = A = x_0
		\end{equation}
	\item<4-> Solution due to initial displacement is then, 
	\begin{align}
		x(t) &= x_0 e^{-\frac{t}{\tau}}   & t>0  \\
		x(t) &= 0  & t<0  
	\end{align}
	\end{itemize}
\end{frame}

\begin{frame}[c]{Homogenous spring-dashpot solution}
	\begin{itemize}
		\item<1->  Solution due to initial displacement is then, 
		\begin{align}
			x(t) &= x_0 e^{-\frac{t}{\tau}}   & t>0  \\
			x(t) &= 0  & t<0  
		\end{align}
	\item<2-> Time constant controls speed at which system returns to equilibrium, $x=0$.
	\begin{itemize}
		\item[-] Big $\tau \rightarrow$ slow return. 
		\item[-] Small $\tau \rightarrow$ fast return.
	\end{itemize}
	\item System returns to equilibrium with \textit{no oscillation}.
	\end{itemize}
\end{frame}

\begin{frame}[c]{Homogenous spring-dashpot solution}
	\begin{itemize}
		\item<1->  Solution due to initial displacement is then, 
		\begin{align}
			x(t) &= x_0 e^{-\frac{t}{\tau}}   & t>0  \\
			x(t) &= 0  & t<0  
		\end{align}
	\end{itemize}

\end{frame}



\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item<1-> Undamped homogenous mass-spring EoM,
				\begin{equation}\label{}
					0 = m\ddot{x} + kx \quad \quad (F_e = c= 0)
				\end{equation}
				\item<2-> Divide through by $m$ to get,
				\begin{equation}\label{}
					0 = \ddot{x} + \frac{k}{m}x =  \ddot{x} + \omega_n^2 x 
				\end{equation}
				\item<3-> Consider solution of the form $x(t) = Ae^{st}$,
				\begin{equation}\label{}
					0 = s^2 Ae^{st} + \omega_n^2 Ae^{st}
				\end{equation}
				\item<4-> Divide through by $Ae^{st}$ (charac. eq.),
				\begin{equation}\label{}
					0 = s^2 + \omega_n^2  \quad\quad s^2 = -\omega_n^2
				\end{equation}
			\end{itemize}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
			\begin{itemize}
				\item<1-> Characteristic equation,
				\begin{equation}\label{}
					0 = s^2 + \omega_n^2  \quad\quad s^2 = -\omega_n^2
				\end{equation}
			\item<2-> Square root of both sides ($i = \sqrt{-1}$), 
			\begin{equation}
				s = \pm \sqrt{-\omega_n^2} = \pm i \omega _n
			\end{equation}
			\item<3-> There are two roots, so the solution is the sum of each,
			\begin{equation}
			x(t) = A_1 e^{i\omega_n t} + A_2 e^{-i\omega_n t}
			\end{equation}
			\item<4-> $A_1$ and $A_2$ are constants that depend on initial conditions, $x_0$ and $\dot{x}_0$.
			\item<4-> Imaginary roots lead to complex exponentials - \textit{harmonic motion at frequency $\omega_n$.}
			\begin{itemize}
				\item[-] We call $\omega_n$ the natural frequency of vibration, or characteristic frequency. 
				\item[-] Force free undamped system will \textit{always} oscillate at $\omega_n$, regardless of IC.
			\end{itemize}
			\end{itemize}
\end{frame}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Undamped homogenous solution, with $\omega_n = \sqrt{k/m}$,
		\begin{equation}
			x(t) = A_1 e^{i\omega_n t} + A_2 e^{-i\omega_n t}
		\end{equation}
%		\item For convenience we have been using complex exponentials - but on physical grounds the solution \textit{must} be real...
		\item<2-> Solve for the constants $A_1$ and $A_2$ in terms of initial displacement and velocity.
		\begin{align}
			x(0) = x_0 &= A_1 + A_2\\
			\dot{x}(0) = \dot{x}_0 &= i\omega_n A_1 -i\omega_n A_2 \quad\rightarrow\quad  \frac{\dot{x}_0}{i\omega_n} = A_1-A_2
		\end{align} 
	\item<3-> From the above we obtain, 
	\begin{equation}
		A_1 = \frac{1}{2}\left(x_0 +\frac{\dot{x}_0}{i\omega}\right) \quad \quad A_2 = \frac{1}{2}\left(x_0 -\frac{\dot{x}_0}{i \omega}\right)
	\end{equation}
	\item<3-> Notice that $A_1$ and$A_2$ are complex conjugates... So only two independent constants since $A_1 = a+ib$ and $A_2=a-ib$.
	\begin{itemize}
		\item[-] This is expected as system is second order - two only two arbitrary constants.
	\end{itemize} 
	\end{itemize}
\end{frame}


\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Substitute the constants into solution and use Euler's formula, 
		\begin{align}
			x(t) &= \frac{1}{2}\left(x_0 +\frac{\dot{x}}{i\omega}\right) e^{i\omega_n t} + \frac{1}{2}\left(x_0 -\frac{\dot{x}}{i\omega}\right) e^{-i\omega_n t} \\
			&=  \frac{1}{2}x_0\underbrace{ \left(e^{i\omega_n t}  + e^{-i\omega_n t} \right) }_{2\cos\omega_n t}+\frac{1}{2}\frac{\dot{x}}{\omega}  \underbrace{\frac{1}{i}\left(e^{i\omega_n t} -e^{-i\omega_n t} \right)}_{2\sin\omega_n t}
		\end{align}
	\item<2-> From the above, the \textit{real} solution is given by,
	\begin{equation}
		x(t) = x_0 \cos\omega_n t + \frac{\dot{x}_0}{\omega_n} \sin\omega_n t
	\end{equation}
	\item<3-> Satisfying initial conditions which are both real, has caused the imaginary part of $x(t)$ to vanish. Expected, true displacement must be real.
	\item<4-> We don't have to go through IC steps to make imaginary part disappear... 
	\end{itemize}
\end{frame}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Alternatively, we can use the solution in its original complex form, $	x(t) = Ae^{i\omega_n t}$, where $A=a+ib$, and take the real part of the response,
		\begin{align}
			\mbox{Real}[x(t)] &= \mbox{Real}[(a+ib)(\cos\omega_n t +i\sin\omega_n t)]\\
			 &=a \cos\omega_n t -b\sin\omega_n t
		\end{align}
	\item<2-> This can be written equivalently as, 
		\begin{align}
		\mbox{Real}[x(t)] = |A| \cos(\omega_nt+\phi)
	\end{align}
		where $|A| = \sqrt{a^2 + b^2}$ and $\phi = \tan^{-1}(b/a)$ are the amplitude and phase, respectively. 
		\item<2-> We can find $|A|$ and $\phi$ directly from IC.
\end{itemize}
\end{frame}


\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> To find $|A|$ and $\phi$ from IC consider solution at $t=0$,
	\begin{align}
		x(0) = x_0&= |A|\cos(\phi)\\
		\dot{x}(0) =\dot{x}_0& = -|A|\omega_n\sin(\phi)
	\end{align}
	\item<2-> Pair of simultaneous equations - solve for $\phi$,
	\begin{equation}
		\frac{ \dot{x}_0}{ x_0} = -\omega_n \frac{\sin\phi}{\cos\phi} = -\omega_n \tan\phi \quad \rightarrow \quad \phi = \tan^{-1}\left(-\frac{1}{\omega_n}\frac{\dot{x}_0}{x_0}\right)
	\end{equation}
	\item<3-> Square and sum $x_0$ and $\dot{x}_0/\omega_n$, then solve for $|A|$,
\begin{align}
	x_0^2 + \frac{\dot{x}_0^2}{\omega_n^2}&= |A|^2\cos^2(\phi)+|A|^2\sin^2(\phi) = |A|^2 \overbrace{(\cos^2(\phi)+\sin^2(\phi))}^{=1}
\end{align}
\item<3-> Amplitude becomes, 
\begin{equation}
	|A| = \sqrt{x_0^2 + \frac{\dot{x}_0^2}{\omega_n^2}}
\end{equation}
\end{itemize}
\end{frame}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Solution of system to arbitrary IC can be written in several ways. 
		\item<1-> In complex exponential form,
		\begin{align}
			x(t) &= A_1 e^{i\omega_n t} + A_2 e^{-i\omega_n t} \quad \mbox{with} \quad 	A_1 = \frac{1}{2}\left(x_0 +\frac{\dot{x}_0}{i\omega}\right) \quad  A_2 = \frac{1}{2}\left(x_0 -\frac{\dot{x}_0}{i \omega}\right)\\
			x(t) &= \mbox{Real}[Ae^{i\omega_n t}] \quad \mbox{with} \quad A = x_0 + i\frac{\dot{x}_0}{\omega_n}
		\end{align}
	 	\item<2-> In explicit $\cos(\quad)$ form,
		\begin{equation}
			x(t) = |A|\cos(\omega_n t + \phi) \quad \mbox{with} \quad A = \sqrt{x_0^2 + \frac{\dot{x}_0^2}{\omega_n^2}}\quad \phi = \tan^{-1}\left(-\frac{1}{\omega_n}\frac{\dot{x}_0}{x_0}\right)
		\end{equation}
		\item<3-> Same result! Just different maths...
	\end{itemize}
\end{frame}

\section{Damped homogenous solution}
\begin{frame}[c]{Damped homogenous mass-spring solution}
	\begin{columns}[T]
		\begin{column}{0.55\textwidth}
			\begin{itemize}
				\item<1-> Damped homogenous mass-spring EoM,
				\begin{equation}\label{}
					0 = m\ddot{x} + kx +c\dot{x} \quad \quad (F_e =0)
				\end{equation}
				\item<2-> Divide through by $m$ to get,
				\begin{equation}\label{}
					0 = \ddot{x} +2\zeta\omega_n \dot{x} + \omega_n^2 x 
				\end{equation}
				using the defined viscous damping factor,
				\begin{equation}
					\zeta = \frac{c}{2m\omega_n}
				\end{equation}
			\item<3-> Assume solution of form $x(t) = Ae^{st}$, 
				\begin{equation}\label{}
					0 = s^2 Ae^{st}  +2\zeta\omega_n s Ae^{st} + \omega_n^2 Ae^{st}
				\end{equation}
			\end{itemize}
		\end{column}
		\begin{column}{0.45\textwidth}
			\begin{figure}
				\input{Diagrams/massSpringLSmod.tikz}
				\caption{Mass-spring-damper system}
			\end{figure}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Divide through by $Ae^{st}$,
			\begin{equation}\label{}
			0 = s^2  +2\zeta\omega_n s  + \omega_n^2 
		\end{equation}
		\item<2-> Quadratic in $s$, solve using quadratic formula (we defined $\zeta$ to make this equation nice!),
		\begin{equation}
			\begin{array}{c} 
				s_1 \\
				s_2\end{array} = \left(-\zeta \pm \sqrt{\zeta^2-1}\right)\omega_n
		\end{equation}
	\item<3-> The nature of the solution depends on the value of $\zeta$!
	\begin{itemize}
		\item[-] If $\zeta=0$ then $s_{1,2}$ are purely imaginary and the solution is that of undamped case.
		\item[-] If $0<\zeta<1$ then $s_{1,2}$ are complex (because of negative in square root) and the solution will become a decaying sinusoid.
		\item[-] If $\zeta=1$ then $s_{1,2}$ are purely real and the solution decays exponentially at the fastest possible rate. 
		\item[-] If $\zeta>1$ then $s_{1,2}$ are purely real and the solution decays exponentially at an increasing slow rate. 
	\end{itemize}
	\end{itemize}
\end{frame}


\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Plug $s_{1,2}$ into our solution $x(t) = Ae^{st} $,
		\begin{equation}
			x(t) = A_1e^{ \left(-\zeta + \sqrt{\zeta^2-1}\right)\omega_n t} + A_2 e^{\left(-\zeta - \sqrt{\zeta^2-1}\right)\omega_n t}
		\end{equation}
	\item<2-> We can factor out constant part of exponential $e^{-\zeta \omega_n t}$
	\begin{equation}
		x(t) = \left(A_1e^{ \sqrt{\zeta^2-1}\omega_n t} + A_2 e^{-\sqrt{\zeta^2-1}\omega_n t}\right)e^{-\zeta\omega_n t}
	\end{equation}
	\item<3-> If $\zeta>1$ (so-called \textit{over-damped}) $x(t)$ decays exponentially, with the precise shaper controlled by $A_1$ and $A_2$.
	\item<4-> If $\zeta=1$ (so-called \textit{critical damping}) can be shown that solution has the form, 
	\begin{equation}
		x(t) = (A_1+tA_2)e^{-\omega_n t}
	\end{equation}
	\item<4-> This occurs when,
	\begin{equation}
		\zeta = 1 = \frac{c}{2m\omega_n} \rightarrow c_{cr} = 2m\omega_n = 2\sqrt{km}
	\end{equation}
	\end{itemize}
\end{frame}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> More interesting is the so-called \textit{under-damped case}, $0<\zeta<1$. In this case it is useful to rewrite $ \sqrt{\zeta^2-1}$ as $i \sqrt{1-\zeta^2}$, 
		\begin{equation}
			x(t) = \left(A_1e^{ i \sqrt{1-\zeta^2}\omega_n t} + A_2 e^{-i \sqrt{1-\zeta^2}\omega_n t}\right)e^{-\zeta\omega_n t}
		\end{equation}
		\item<2-> We define the \textit{frequency of damped natural vibration}, $ \omega_d = \sqrt{1-\zeta^2}\omega_n $. 
		\begin{itemize}
			\item[-] The factor of  $\sqrt{1-\zeta^2}$ slightly lowers the system's natural frequency. 
			\item[-] Exponential decay remains proportional to the undamped natural frequency $\omega_n$.
		\end{itemize}
		\item<3-> Using damped natural frequency, our solution has the form,
				\begin{equation}
			x(t) = \left(A_1e^{ i\omega_d t} + A_2 e^{-i \omega_d t}\right)e^{-\zeta\omega_n t}
		\end{equation}
		\item<3-> Note that the bracketed term is identical in form to what we had for the undamped case, just with $\omega_n\rightarrow\omega_d$.
	\end{itemize}
\end{frame}

\begin{frame}[c]{Undamped homogenous mass-spring solution}
	\begin{itemize}
		\item<1-> Reusing the previous \textit{undamped} result, our \textit{damped} solution becomes, 
			\begin{equation}
				x(t) = A\cos(\omega_d t + \phi) e^{-\zeta\omega_n t} 
			\end{equation}
		with the initial conditions, 
		\begin{equation}
			A = \sqrt{x_0^2 + \frac{\dot{x}_0^2}{\omega_d^2}}\quad \mbox{and} \quad \phi = \tan^{-1}\left(-\frac{1}{\omega_d}\frac{\dot{x}_0}{x_0}\right)
		\end{equation}
		\item<2-> We can start an oscillation with an initial displacement and/or velocity. How else might we push the system into motion?
		\begin{itemize}
			\item[-] We might apply a very short force input, i.e. give it a kick...
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}[c]{Unit impulse response}
	\begin{itemize}
		\item<1-> Rather than setting mass into motion by displacing or applying an initial velocity, suppose instead we give it a \textit{kick}.
		\item <2->This kick can be modelled as a constant force $F$ over the infinitesimal time $0<t<\epsilon$.
		\item<3-> During this kick, the velocity of the mass increases from 0 to $\dot{x}(\epsilon)$, after which its motion is governed by the system dynamics.
		\item<4-> From Newton's second law $F=m\ddot{x}$, and simple kinematics $\dot{x}(t) = t\ddot{x}$, we have that the velocity at $t=\epsilon$ is, 
		\begin{equation}
			\dot{x}(\epsilon) = \epsilon\ddot{x} = \frac{\epsilon F}{m}
		\end{equation}
		\item<5-> The product $\epsilon F$ is known as the \textit{impulse} - describes the change in momentum.
		\item<6-> We see that a \textit{unit impulse} ($\epsilon F = 1$) is equivalent to an initial velocity that is inversely proportional to the system mass, $\dot{x}_0 = 1/m$.
		
	\end{itemize}
\end{frame}


\section{Estimating damping}

\begin{frame}[c]{Estimate damping - log-decrement method}
	\begin{itemize}
		\item<1-> The amount of damping in a system is often unknown - it can be estimated from a response measurement $x(t)$. 
		\item Assuming a viscous damping, the response decays exponentially,
		\begin{equation}
			x(t) = A\cos(\omega_d t + \phi) e^{-\zeta\omega_n t} 
		\end{equation}
	\item<2-> Consider amplitude ratio after a period $T$.
	\end{itemize}
\vspace{0.25cm}
	\begin{figure}
	\input{Diagrams/logdecrement.tikz}
\end{figure}
\end{frame}

\begin{frame}[c]{Estimate damping - log-decrement method}
	\begin{itemize}
		\item<1-> Consider amplitude ratio after a period $T$.
		\begin{equation}
			\frac{x_1}{x_2} = \frac{A\cos(\omega_d t_1 + \phi) e^{-\zeta\omega_n t_1} }{A\cos(\omega_d t_2 + \phi) e^{-\zeta\omega_n t_2} }
		\end{equation}
	\item<2-> Note that $t_2 = t_1 + T = t_1 + 2\pi/\omega_d$, and so, 
	\begin{equation}
		\cos(\omega_d t_2 + \phi) = \cos(\omega_d (t_1 + 2\pi/\omega_d) + \phi) = \cos(\omega_d t_1 + 2\pi + \phi) = \cos(\omega_d t_1 + \phi)
	\end{equation}
	\item<3-> Note that the $\cos$ term does not change after $T$...
		\begin{equation}
		\frac{x_1}{x_2} = \frac{e^{-\zeta\omega_n t_1} }{e^{-\zeta\omega_n (t_1+T)} } = e^{\zeta\omega_n T}
	\end{equation}
	\item<4-> We define the so-called \textit{log-decrement}, $\delta = \ln(x_1/x_2)$. This is something we can obtain from measurement.
	\end{itemize}
\end{frame}

\begin{frame}[c]{Estimate damping - log-decrement method}
	\begin{itemize}
		\item<1-> The log-decrement, 
		\begin{equation}
			\delta = \ln(x_1/x_2) = \zeta\omega_n T = \zeta\omega_n \frac{2\pi }{\omega_d} = \frac{\zeta 2\pi}{\sqrt{1-\zeta^2}}
		\end{equation}
	\item<2-> Now we can solve for $\zeta$ to obtain,
	\begin{equation}
		\zeta = \frac{\delta}{\sqrt{(2\pi)^2 + \delta^2}}
	\end{equation}
	\item<3-> Note that for multiple periods, 
	\begin{align}
		\frac{x_1}{x_n} = \frac{x_1}{x_2}\, \frac{x_2}{x_3}  \, \frac{x_3}{x_4} \cdots \frac{x_{n-1}}{x_n} = (e^{\zeta\omega_nT})^n= e^{n\zeta\omega_nT}
	\end{align}
	\item<3-> The log-decrement (and so also $\zeta$) can then be obtained using,
	\begin{equation}
		\delta = \frac{1}{n} \ln(x_1/x_2)
	\end{equation}
	\end{itemize}
\end{frame}
